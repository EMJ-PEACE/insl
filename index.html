Source code 
#include <stdio.h>
#include <stdlib.h>
#include <omp.h>

int main() {
    int n;

    printf("Enter the size of the matrix/vector (n): ");
    scanf("%d", &n);

    double **A = malloc(n * sizeof(double *));
    double *x = malloc(n * sizeof(double));
    double *y = malloc(n * sizeof(double));

    printf("Enter elements of matrix A (%d x %d):\n", n, n);
    for (int i = 0; i < n; i++) {
        A[i] = malloc(n * sizeof(double));
        for (int j = 0; j < n; j++) scanf("%lf", &A[i][j]);
    }

    printf("Enter elements of vector x (%d elements):\n", n);
    for (int i = 0; i < n; i++) scanf("%lf", &x[i]);

    #pragma omp parallel for
    for (int i = 0; i < n; i++) {
        y[i] = 0;
        for (int j = 0; j < n; j++) y[i] += A[i][j] * x[j];
    }

    printf("Result vector y = A * x:\n");
    for (int i = 0; i < n; i++) printf("%lf ", y[i]);
    printf("\n");

    for (int i = 0; i < n; i++) free(A[i]);

    free(A);
    free(x);
    free(y);

    return 0;
}

#include <stdio.h>
#include <omp.h>

int main() {
    double clothing[] = {1200.5, 799.0, 450.0};
    double gaming[] = {3000.0, 1500.0};
    double grocery[] = {250.5, 130.75, 90.0, 60.25};
    double stationary[] = {45.5, 99.0};

    int n_clothing = sizeof(clothing) / sizeof(clothing[0]);
    int n_gaming = sizeof(gaming) / sizeof(gaming[0]);
    int n_grocery = sizeof(grocery) / sizeof(grocery[0]);
    int n_stationary = sizeof(stationary) / sizeof(stationary[0]);

    double total = 0;

    #pragma omp parallel for reduction(+:total)
    for (int i = 0; i < n_clothing; i++) total += clothing[i];

    #pragma omp parallel for reduction(+:total)
    for (int i = 0; i < n_gaming; i++) total += gaming[i];

    #pragma omp parallel for reduction(+:total)
    for (int i = 0; i < n_grocery; i++) total += grocery[i];

    #pragma omp parallel for reduction(+:total)
    for (int i = 0; i < n_stationary; i++) total += stationary[i];

    printf("Final Bill Amount: %.2lf\n", total);

    return 0;
}
PROGRAM 03

Calculation of Pi Using OpenMP Sections

OpenMP
Compile and run on Linux 
gcc -fopenmp program3.c -o program3
./program3
Source code 
#include <stdio.h>
#include <omp.h>

int main() {
    long num_steps = 100000000;
    double step = 1.0 / num_steps;

    double sum1 = 0, sum2 = 0;
    double pi;

    double start = omp_get_wtime();

    #pragma omp parallel sections
    {
        #pragma omp section
        {
            for (long i = 0; i < num_steps / 2; i++) {
                double x = (i + 0.5) * step;
                sum1 += 4.0 / (1.0 + x * x);
            }
        }

        #pragma omp section
        {
            for (long i = num_steps / 2; i < num_steps; i++) {
                double x = (i + 0.5) * step;
                sum2 += 4.0 / (1.0 + x * x);
            }
        }
    }

    pi = step * (sum1 + sum2);

    double end = omp_get_wtime();

    printf("Computed Pi = %.15f\n", pi);
    printf("Execution Time = %f seconds\n", end - start);

    return 0;
}

Fibonacci Series Using OpenMP Barrier

OpenMP
Compile and run on Linux 
gcc -fopenmp program4.c -o program4

Source code 
#include <stdio.h>
#include <omp.h>

#define MAX 100

int main() {
    int n;
    int fib[MAX];

    printf("Enter the number of terms in Fibonacci Series: ");
    scanf("%d", &n);

    if (n <= 0 || n > MAX) {
        printf("Invalid input. Please enter a number between 1 and %d.\n", MAX);
        return 1;
    }

    #pragma omp parallel num_threads(2)
    {
        if (omp_get_thread_num() == 0) {
            fib[0] = 0;
            if (n > 1) fib[1] = 1;

            for (int i = 2; i < n; i++)
                fib[i] = fib[i - 1] + fib[i - 2];
        }

        #pragma omp barrier

        if (omp_get_thread_num() == 1) {
            printf("Fibonacci Series:\n");
            for (int i = 0; i < n; i++) printf("%d ", fib[i]);
            printf("\n");
        }
    }

    return 0;
}
PROGRAM 05

Gold Medalist — Highest CGPA Using OpenMP

OpenMP
Compile and run on Linux

Source code 
#include <stdio.h>
#include <omp.h>

#define MAX 100

struct Student {
    char name[50];
    float cgpa;
};

int main() {
    int n;
    struct Student students[MAX];

    printf("Enter number of students: ");
    scanf("%d", &n);

    if (n <= 0 || n > MAX) {
        printf("Invalid number of students.\n");
        return 1;
    }

    for (int i = 0; i < n; i++) {
        printf("Enter name and CGPA of student %d: ", i + 1);
        scanf("%49s %f", students[i].name, &students[i].cgpa);
    }

    float maxCGPA = -1;
    int maxIndex = -1;

    #pragma omp parallel for
    for (int i = 0; i < n; i++) {
        #pragma omp critical
        {
            if (maxIndex == -1 || students[i].cgpa > maxCGPA ||
                (students[i].cgpa == maxCGPA && i < maxIndex)) {
                maxCGPA = students[i].cgpa;
                maxIndex = i;
            }
        }
    }

    printf("\n Gold Medalist: %s with CGPA %.2f\n",
           students[maxIndex].name, students[maxIndex].cgpa);

    return 0;
}
PROGRAM 06

Sequential and Parallel Matrix Multiplication

OpenMP
Compile and run on Linux 
gcc -fopenmp program6.c -o program6
./program6
Source code 
#include <stdio.h>
#include <stdlib.h>
#include <omp.h>

float** allocateMatrix(int n) {
    float **a = malloc(n * sizeof(float *));
    for (int i = 0; i < n; i++) a[i] = malloc(n * sizeof(float));
    return a;
}

void fillMatrix(float **a, int n) {
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            a[i][j] = rand() % 10;
}

void sequentialMultiply(float **A, float **B, float **C, int n) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            C[i][j] = 0;
            for (int k = 0; k < n; k++)
                C[i][j] += A[i][k] * B[k][j];
        }
    }
}

void parallelMultiply(float **A, float **B, float **C, int n) {
    #pragma omp parallel for collapse(2)
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            C[i][j] = 0;
            for (int k = 0; k < n; k++)
                C[i][j] += A[i][k] * B[k][j];
        }
    }
}

void freeMatrix(float **a, int n) {
    for (int i = 0; i < n; i++) free(a[i]);
    free(a);
}

int main() {
    int n;

    printf("Enter matrix size (1000, 2000, 3000): ");
    scanf("%d", &n);

    if (n <= 0 || n > 3000) {
        printf("Invalid matrix size.\n");
        return 1;
    }

    float **A = allocateMatrix(n);
    float **B = allocateMatrix(n);

    float **C_seq = allocateMatrix(n);
    float **C_par = allocateMatrix(n);

    fillMatrix(A, n);
    fillMatrix(B, n);

    double start, end;

    start = omp_get_wtime();
    sequentialMultiply(A, B, C_seq, n);
    end = omp_get_wtime();

    printf("Sequential Time: %.4f seconds\n", end - start);

    start = omp_get_wtime();
    parallelMultiply(A, B, C_par, n);
    end = omp_get_wtime();

    printf("Parallel Time (OpenMP): %.4f seconds\n", end - start);

    freeMatrix(A, n);
    freeMatrix(B, n);
    freeMatrix(C_seq, n);
    freeMatrix(C_par, n);

    return 0;
}

Blocking and Non-Blocking Communication Using MPI

Source code 
#include <stdio.h>
#include <mpi.h>

int main(int argc, char *argv[]) {
    int rank, size;
    int data_send = 100;
    int data_recv;

    MPI_Request request;
    MPI_Status status;

    MPI_Init(&argc, &argv);

    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (size < 2) {
        if (rank == 0)
            printf("Run the program with at least 2 processes.\n");

        MPI_Finalize();
        return 0;
    }

    if (rank == 0) {
        printf("[Blocking] Process 0 sending data: %d\n", data_send);
        MPI_Send(&data_send, 1, MPI_INT, 1, 0, MPI_COMM_WORLD);
    }
    else if (rank == 1) {
        MPI_Recv(&data_recv, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &status);
        printf("[Blocking] Process 1 received data: %d\n", data_recv);
    }

    MPI_Barrier(MPI_COMM_WORLD);

    if (rank == 0) {
        data_send = 200;
        printf("[Non-Blocking] Process 0 initiating send: %d\n", data_send);
        MPI_Isend(&data_send, 1, MPI_INT, 1, 1, MPI_COMM_WORLD, &request);
        MPI_Wait(&request, &status);
    }
    else if (rank == 1) {
        MPI_Irecv(&data_recv, 1, MPI_INT, 0, 1, MPI_COMM_WORLD, &request);
        MPI_Wait(&request, &status);
        printf("[Non-Blocking] Process 1 received data: %d\n", data_recv);
    }

    MPI_Finalize();
    return 0;
}

#include <stdio.h>
#include <mpi.h>

int main(int argc, char *argv[]) {
    int rank, size;

    int numbers[4] = {10, 20, 30, 40};

    int recv_num;
    int doubled;
    int gathered[4];
    int total_sum;

    MPI_Init(&argc, &argv);

    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (size != 4) {
        if (rank == 0)
            printf("Please run this with exactly 4 processes.\n");
        MPI_Finalize();
        return 1;
    }

    int broadcast_value = 100;

    MPI_Bcast(&broadcast_value, 1, MPI_INT, 0, MPI_COMM_WORLD);

    printf("Process %d received broadcast value: %d\n", rank, broadcast_value);

    MPI_Scatter(numbers, 1, MPI_INT, &recv_num, 1, MPI_INT, 0, MPI_COMM_WORLD);

    printf("Process %d received number: %d\n", rank, recv_num);

    doubled = recv_num * 2;

    MPI_Gather(&doubled, 1, MPI_INT, gathered, 1, MPI_INT, 0, MPI_COMM_WORLD);

    MPI_Reduce(&doubled, &total_sum, 1, MPI_INT, MPI_SUM, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        printf("Root gathered doubled numbers: ");
        for (int i = 0; i < 4; i++) printf("%d ", gathered[i]);
        printf("\n");
        printf("Total sum of doubled numbers: %d\n", total_sum);
    }

    MPI_Bcast(&total_sum, 1, MPI_INT, 0, MPI_COMM_WORLD);

    printf("Process %d sees total sum as: %d\n", rank, total_sum);

    MPI_Finalize();

    return 0;
}

MPI Squaring Numbers and Finding Sum of Squares

Source code 
#include <stdio.h>
#include <mpi.h>

int main(int argc, char *argv[]) {
    int rank, size;

    int n = 4;
    int data[4] = {1, 2, 3, 4};

    int recv;
    int squared;
    int gathered[4];
    int total_sum;

    MPI_Init(&argc, &argv);

    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (size != n) {
        if (rank == 0) printf("Run with exactly %d processes.\n", n);
        MPI_Finalize();
        return 0;
    }

    MPI_Bcast(&n, 1, MPI_INT, 0, MPI_COMM_WORLD);

    MPI_Scatter(data, 1, MPI_INT, &recv, 1, MPI_INT, 0, MPI_COMM_WORLD);

    squared = recv * recv;

    MPI_Gather(&squared, 1, MPI_INT, gathered, 1, MPI_INT, 0, MPI_COMM_WORLD);

    MPI_Reduce(&squared, &total_sum, 1, MPI_INT, MPI_SUM, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        printf("Squared values: ");
        for (int i = 0; i < n; i++) printf("%d ", gathered[i]);
        printf("\nSum of squares = %d\n", total_sum);
    }

    MPI_Finalize();

    return 0;
}
